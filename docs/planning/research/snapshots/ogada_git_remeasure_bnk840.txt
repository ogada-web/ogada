```js
export async function validateProgramSchedulePhotoFile(file) {
  if (!file) return "업로드할 프로그램 사진 파일이 없습니다.";
  if (file.size > PROGRAM_SCHEDULE_PHOTO_MAX_BYTES)
    return "프로그램 사진 파일 크기는 5MB 이하여야 합니다.";
  const contentType = normalizeProgramSchedulePhotoContentType(file.type);
  if (!contentType || !PROGRAM_SCHEDULE_PHOTO_ALLOWED_TYPES.has(contentType))
    return PROGRAM_SCHEDULE_PHOTO_UNSUPPORTED_FORMAT_MESSAGE;
  let header;
  try {
    header = await readFileHeaderBytes(file, PROGRAM_SCHEDULE_PHOTO_MAGIC_HEADER_BYTES);
  } catch {
    return PROGRAM_SCHEDULE_PHOTO_UNSUPPORTED_FORMAT_MESSAGE;
  }
  if (!matchesProgramSchedulePhotoMagicBytes(header, contentType))
    return PROGRAM_SCHEDULE_PHOTO_UNSUPPORTED_FORMAT_MESSAGE;
  return null;
}
```
