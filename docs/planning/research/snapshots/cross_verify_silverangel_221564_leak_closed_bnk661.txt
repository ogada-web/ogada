`utils/clientListFilters.js`:
```js
export const CLIENT_LIST_BRANCH_QUERY_PARAM = "branchId";   // NEW

export function buildClientListSearchHref(query, branchId) {  // ← 시그니처에 branchId 추가
  const trimmed = String(query ?? "").trim();
  const branch = String(branchId ?? "").trim();
  const params = new URLSearchParams();
  if (branch) {
    params.set(CLIENT_LIST_BRANCH_QUERY_PARAM, branch);
  }
  if (trimmed) {
    params.set(CLIENT_LIST_SEARCH_QUERY_PARAM, trimmed);
  }
  const queryString = params.toString();
  return queryString ? `/clients?${queryString}` : "/clients";
}

export function readClientListBranchFromQuery(searchParams) {  // NEW
  const raw =
    typeof searchParams?.get === "function"
      ? searchParams.get(CLIENT_LIST_BRANCH_QUERY_PARAM)
      : "";
  return String(raw ?? "").trim() || CLIENT_LIST_FILTER_ALL;
}
```
