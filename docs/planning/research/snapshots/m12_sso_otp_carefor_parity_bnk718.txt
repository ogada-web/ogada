function open_sujifine()
{
    if('1' == '1')
    {
        alert("체험하기에서는 재무회계 서비스 접근이 불가능합니다.");
        return false;
    }

    var f_sujifine = $("#openSujifineForm");

    $(f_sujifine).find("input[name=usmusid]").val('');
    $(f_sujifine).find("input[name=otp]").val('');

    $.ajax({
        type: "POST",
        url: "/share/exe/sujifine/get.sujifine_id.exe.php",
        data: {stmmgno: '588'},
        async: false,
        success: function(data){
            data = $.parseJSON( data.replace(/\'/g, "\"") );

            if(data.usmusid != '')
            {
                window.open('', "sujifineLogin", "width=1920,height=935,resizable=no,status=no,directories=no,location=no,menubar=no");

                $(f_sujifine).find("input[name=usmusid]").val(data.usmusid);
                $(f_sujifine).find("input[name=otp]").val(data.otp);

                $(f_sujifine).submit();
            }
            else
            {
                alert("재무회계 서비스에 등록되지 않은 사용자입니다.");
            }
        }
    });
}

function linkInzi()
{
    $.post("/share/exe/inzi/link.inzi.exe.php", {}, function(data1) {
    	  $.post("/share/exe/inzi/login.inzi.exe.php", {}, function(data2) {
            var param = $.parseJSON( data2 );
            openInzi(param);
        });
    });
}

function openInzi(param)
{
    var thisForm = $("#openInziFrm");
    thisForm.find("input[name=center_mgno]").val( param['center_mgno'] );
    thisForm.find("input[name=center_manager_yn]").val( param['center_manager_yn'] );
    thisForm.find("input[name=auth_key]").val( param['auth_key'] );
    thisForm.submit();
}
</script>
